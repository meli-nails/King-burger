Este paso lo hace una sola vez el propietario.insert into public.admins (user_id, role, is_active) select id, 'owner', true from auth.users where email = 'TU_CORREO@EJEMPLO.COM' on conflict (user_id) do update set role = 'owner', is_active = true;
